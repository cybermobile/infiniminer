servername = Unnamed Server
maxplayers = 16

# Before making your server public, MAKE SURE THAT PORT 5565 (UDP) IS ACCESSIBLE FROM THE OUTSIDE OF YOUR NETWORK!
# For most users, this means setting the port to be forwarded by your router through your router control interface.
public = false

# The port to listen on (UDP). LAN discovery only finds servers on the default port, 5565.
port = 5565

# Public servers are announced to this server list URL. The original list (apps.keithholman.net) is gone, so it
# is empty by default. Example: serverlist = http://example.com/post
serverlist = 

# This value specifies the amount of cash required for a team to win the game.
winningcash = 10000

# cash: the original game, a race to bank treasure. hill: king of the hill, as well as banking treasure. A metal
# platform in the middle of the map (with ramps up to it) earns $25 a second for the team with more players on it.
gamemode = cash

# This flag specifies whether or not generated levels will include lava.
includelava = true

# This value specifies the number of ore veins found in the world.
orefactor = 10

# If this flag is true, all blocks will be free and the game will be unwinnable.
sandbox = false

# If this flag is true, each team gets a home base (a crater with a metal floor) in opposite corners
# of the map, and players spawn there.
homebases = false

# If this flag is true, shock blocks kill anyone who touches them, not just anyone hitting their underside.
shocktouch = false

# Match length in minutes (0 = no limit). When time runs out the richer team wins; a tie goes to overtime,
# where the next deposit wins.
timelimit = 0

# Extra cash needed to win for every connected player, so bigger games last longer.
winningcashperplayer = 0

# If this flag is true, players cannot join a team that already has more players than the other.
autobalance = false

# The size of the world cube: 64 (the original), 96 or 128. Bigger worlds have proportionally more ore,
# gold and diamonds.
mapsize = 64

# Players who type "/login <password>" in chat can run any console command from chat (e.g. /kick Name).
# Leave empty to disable remote admin.
adminpassword = 

# Computer players (0 to 16). They mine for their team, keep the teams even, and leave to make room
# when people join a full server. Change it while running with the "bots <count>" console command.
bots = 0

# How well bots play: easy, normal or hard. Even hard bots dig and walk no faster than players.
botskill = normal

# Ask your router to forward the server's UDP port (UPnP), so players on the internet can join. If the
# router has UPnP switched off, forward the port by hand instead. The console says which happened.
upnp = false

# Keep the world (and the match score and clock) in world.sav, so stopping and starting the server carries
# on where it was. Saved every two minutes and on shutdown; a new world is made once a match ends.
persist = false

# 0 makes a different world every time; any other number always makes the same world for the same size.
seed = 0
